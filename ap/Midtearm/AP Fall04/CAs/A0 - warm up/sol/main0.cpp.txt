#include <iostream>
#include <sstream>
#include <string>

int convert(int raw_value, const std::string& from, const std::string& to) {
    if (from == "Meter" && to == "Centimeter") {
        return raw_value * 100;
    }
    if (from == "Kilogram" && to == "Gram") {
        return raw_value * 1000;
    }
    if (from == "Second" && to == "Hour") {
        return raw_value / 3600;
    }
    if (from == "Celsius" && to == "Fahrenheit") {
        return raw_value * 1.8 + 32;
    }
    return 0;
}

int main() {
    std::string line;
    while (std::getline(std::cin, line)) {
        std::stringstream ss(line);
        int raw_value;
        std::string from, to;
        ss >> raw_value >> from >> to;
        std::cout << convert(raw_value, from, to) << std::endl;
    }

    return 0;
}
