#include <algorithm>
#include <fstream>
#include <iostream>
#include <set>
#include <string>
#include <unordered_map>
#include <vector>

using namespace std;

const int FAVORITE_SONG_SCORE = 5;

namespace commands {
const string LISTEN_SONG = "listen_song";
const string CREATE_PLAYLIST = "create_playlist";
const string LISTEN_PLAYLIST = "listen_playlist";
const string RECOMMEND_SONG = "recommend_song";
}; // namespace commands

struct Song {
    string name;
    int id;
    int year;
    string genre;
    string artist;
};

struct Playlist {
    string name;
    vector<Song*> songs;
};

struct User {
    string username;
    unordered_map<int, int> musicHistory;
    string favoriteGenre;
    vector<Playlist*> playlists;
};

struct SongItem {
    int score;
    Song* song;
};

vector<string> tokenize(const string& str, const string delimiter) {
    vector<string> tokenized;
    size_t start = 0, end = str.find(delimiter);
    if (end == string::npos) {
        tokenized.push_back(str);
        return tokenized;
    }
    while (true) {
        tokenized.push_back(str.substr(start, end - start));
        start = end + 1;
        end = str.find(delimiter, start);
        if (end == string::npos) {
            tokenized.push_back(str.substr(start));
            break;
        }
    }
    return tokenized;
}

unordered_map<int, int> parseMusicHistory(const string& historyStr) {
    unordered_map<int, int> musicHistory;
    vector<string> songEnteries = tokenize(historyStr, ";");
    for (const auto& entry : songEnteries) {
        vector<string> parts = tokenize(entry, ":");
        int songId = stoi(parts[0]);
        int listenCount = stoi(parts[1]);
        musicHistory[songId] = listenCount;
    }

    return musicHistory;
}

inline Song* parseSongLine(const vector<string>& tokens) {
    return new Song{
        tokens[0],
        stoi(tokens[1]),
        stoi(tokens[2]),
        tokens[3],
        tokens[4]};
}

inline User* parseUserLine(const vector<string>& tokens) {
    return new User{
        tokens[0],
        parseMusicHistory(tokens[1]),
        tokens[2],
        {}};
}

void loadCSVData(char* argv[], vector<Song*>& songs_, vector<User*>& users_) {
    for (int i = 1; i < 3; i++) {
        string csvPath = argv[i];
        ifstream inputFile(csvPath);
        string csvLine;
        getline(inputFile, csvLine);
        while (getline(inputFile, csvLine)) {
            vector<string> tokenizedLine = tokenize(csvLine, ",");
            if (i == 1)
                songs_.push_back(parseSongLine(tokenizedLine));
            else
                users_.push_back(parseUserLine(tokenizedLine));
        }
    }
}

inline Song* findSongById(int songId_, vector<Song*>& songs_) {
    return *find_if(songs_.begin(), songs_.end(), [songId_](const Song* song) { return song->id == songId_; });
}

inline User* findUserByUsername(string username, vector<User*>& users_) {
    return *find_if(users_.begin(), users_.end(), [username](const User* user) { return user->username == username; });
}

inline Playlist* findPlaylistByName(string playlistName, vector<Playlist*>& playlists) {
    return *find_if(playlists.begin(), playlists.end(), [playlistName](const Playlist* playlist) { return playlist->name == playlistName; });
}

int getSongScore(Song* song, User* user, vector<Song*>& songs_) {
    int score = 0;

    for (const auto& [songId, songCount] : user->musicHistory) {
        Song* currentSong = findSongById(songId, songs_);
        if (currentSong->artist == song->artist)
            score += songCount;
        if (currentSong->year == song->year)
            score += songCount;
        if (currentSong->genre == song->genre)
            score += songCount;
    }
    if (user->favoriteGenre == song->genre)
        score += FAVORITE_SONG_SCORE;

    return score;
}

vector<Song*> selectSongsbyScore(vector<SongItem> songScores_, int count_, unordered_map<int, int>& musicHistory) {
    sort(songScores_.begin(), songScores_.end(), [](const SongItem i1, const SongItem i2) { return (i1.score == i2.score) ? (i1.song->name < i2.song->name) : (i1.score > i2.score); });
    vector<Song*> selectedSongs;
    for (SongItem songItem : songScores_) {
        if (size(selectedSongs) == count_)
            break;
        if (musicHistory.find(songItem.song->id) == musicHistory.end())
            selectedSongs.push_back(songItem.song);
    }

    return selectedSongs;
}

void listenSong(string username, int songId, vector<Song*>& songs_, vector<User*>& users_) {
    auto songIt = std::find_if(songs_.begin(), songs_.end(), [songId](const Song* s) { return s->id == songId; });
    string songName = (*songIt)->name;

    auto userIt = std::find_if(users_.begin(), users_.end(), [username](const User* u) { return u->username == username; });
    (*userIt)->musicHistory[songId]++;

    cout << username << " listened to " << songName << "!\n";
}

void createPlaylist(string playlistName, string username, int songCount, vector<Song*>& songs_, vector<User*>& users_) {
    User* user = findUserByUsername(username, users_);
    vector<SongItem> songScores;
    for (Song* song : songs_) {
        songScores.push_back({getSongScore(song, user, songs_), song});
    }
    Playlist* newPlaylist = new Playlist{playlistName, selectSongsbyScore(songScores, songCount, user->musicHistory)};
    user->playlists.push_back(newPlaylist);

    cout << newPlaylist->name << ":\n";
    for (Song* song : newPlaylist->songs) {
        cout << song->name << " " << song->artist << " " << song->year << " " << song->genre << '\n';
    }
}

void listenPlaylist(string username, string playlistName, string creatorUsername, vector<Song*>& songs_, vector<User*>& users_) {
    User* creatorUser = findUserByUsername(creatorUsername, users_);
    User* user = findUserByUsername(username, users_);
    Playlist* playlist = findPlaylistByName(playlistName, creatorUser->playlists);
    for (Song* song : playlist->songs) {
        if (user->musicHistory.find(song->id) != user->musicHistory.end())
            user->musicHistory[song->id]++;
        else
            user->musicHistory[song->id] = 1;
    }

    cout << username << " listened to the " << playlistName << " playlist!\n";
}

void recommendSong(string username_, vector<Song*>& songs_, vector<User*>& users_) {
    User* user = findUserByUsername(username_, users_);
    vector<SongItem> songScores;
    for (Song* song : songs_) {
        songScores.push_back({getSongScore(song, user, songs_), song});
    }
    Song* selectedSong = selectSongsbyScore(songScores, 1, user->musicHistory)[0];

    cout << "Recommended song for " << username_ << ":\n";
    cout << selectedSong->name << ' ' << selectedSong->artist << ' ' << selectedSong->year << ' ' << selectedSong->genre << '\n';
}

void commandHandler(vector<Song*>& songs_, vector<User*>& users_) {
    string commandLine;
    while (getline(cin, commandLine)) {
        if (commandLine == "") break;
        vector<string> tokenizedCommandLine = tokenize(commandLine, " ");
        string command = tokenizedCommandLine[0];
        if (command == commands::LISTEN_SONG) {
            string username = tokenizedCommandLine[1];
            int songId = stoi(tokenizedCommandLine[2]);
            listenSong(username, songId, songs_, users_);
        }
        else if (command == commands::CREATE_PLAYLIST) {
            string playlistName = tokenizedCommandLine[1];
            string username = tokenizedCommandLine[2];
            int songCount = stoi(tokenizedCommandLine[3]);
            createPlaylist(playlistName, username, songCount, songs_, users_);
        }
        else if (command == commands::LISTEN_PLAYLIST) {
            string username = tokenizedCommandLine[1];
            string playlistName = tokenizedCommandLine[2];
            string creatorUsername = tokenizedCommandLine[3];
            listenPlaylist(username, playlistName, creatorUsername, songs_, users_);
        }
        else if (command == commands::RECOMMEND_SONG) {
            string username = tokenizedCommandLine[1];
            recommendSong(username, songs_, users_);
        }
    }
}

void sutify(char* argv[]) {
    vector<Song*> songs;
    vector<User*> users;
    loadCSVData(argv, songs, users);
    commandHandler(songs, users);
}

int main(int argc, char* argv[]) {
    if (argc != 3) {
        cerr << "Invalid number of attributes entered!, Usage: ./Sutify ./path/to/songs/file ./path/to/users/file";
        return EXIT_FAILURE;
    }

    sutify(argv);
    return 0;
}
