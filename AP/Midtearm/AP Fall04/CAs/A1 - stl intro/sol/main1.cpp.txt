#include <deque>
#include <iostream>
#include <map>
#include <unordered_map>
#include <sstream>
#include <string>

namespace commands {
const std::string ADD_PRODUCT = "add_product";
const std::string INCREASE_QUANTITY = "increase_quantity";
const std::string ADD_ORDER = "add_order";
const std::string PROCESS_ORDERS = "process_orders";
const std::string REPORT_PENDING = "report_pending";
const std::string SOLD_QUANTITY = "sold_quantity";
const std::string SHOW_STOCK = "show_stock";
} 

struct Order {
    std::string customer;
    std::string product;
    uint quantity;
};

struct Warehouse {
    std::map<std::string /*product*/, uint /*quantity*/> productStock;
    std::unordered_map<std::string /*product*/, uint /*quantity*/> soldProducts;
    std::deque<Order> pendingOrders;
    std::deque<Order> failedOrders;
    std::deque<Order> completedOrders;
};

void addProduct(Warehouse& warehouse, const std::string& product, uint quantity) {
    auto itr = warehouse.productStock.find(product);
    if (itr == warehouse.productStock.end()) {
        warehouse.productStock[product] = quantity;
        return;
    }

    itr->second += quantity;
}

void increaseQuantity(Warehouse& warehouse, const std::string& product, uint quantity) {
    warehouse.productStock[product] += quantity;
}

void addOrder(Warehouse& warehouse, const std::string& customer, const std::string& product, uint quantity) {
    warehouse.pendingOrders.push_front(Order{customer, product, quantity});
}

void processOrders(Warehouse& warehouse) {
    while (!warehouse.pendingOrders.empty()) {
        Order order = warehouse.pendingOrders.back();
        warehouse.pendingOrders.pop_back();
        auto itr = warehouse.productStock.find(order.product);
        if (itr == warehouse.productStock.end() || itr->second < order.quantity) {
            warehouse.failedOrders.push_front(order);
            continue;
        }

        itr->second -= order.quantity;
        warehouse.completedOrders.push_front(order);
        if (warehouse.soldProducts.find(order.product) == warehouse.soldProducts.end()) {
            warehouse.soldProducts[order.product] = 0;
        }
        warehouse.soldProducts[order.product] += order.quantity;
    }
}

void reportPending(const Warehouse& warehouse) {
    if (warehouse.pendingOrders.empty()) {
        std::printf("No pending orders\n");
        return;
    }

    int orderNumber = 1;
    for (auto itr = warehouse.pendingOrders.rbegin(); itr != warehouse.pendingOrders.rend(); itr++) {
        const Order& order = *itr;
        std::printf("%d. %s: %s %d\n", orderNumber, order.customer.c_str(), order.product.c_str(), order.quantity);
        orderNumber++;
    }
}

void reportSoldQuantity(const Warehouse& warehouse, const std::string& product) {
    auto itr = warehouse.soldProducts.find(product);
    if (itr == warehouse.soldProducts.end()) {
        std::printf("0\n");
        return;
    }

    std::printf("%d\n", itr->second);
}

void showStock(const Warehouse& warehouse) {
    if (warehouse.productStock.empty()) {
        std::printf("No products found\n");
        return;
    }
    
    int productNumber = 1;
    for (auto [product, quantity] : warehouse.productStock) {
        std::printf("%d. %s: %d\n", productNumber, product.c_str(), quantity);
        productNumber++;
    }
}

void processCommand(Warehouse& warehouse, const std::string& line) {
    std::stringstream ss(line);
    std::string command;
    ss >> command;
    if (command == commands::ADD_PRODUCT) {
        std::string product;
        uint quantity;
        ss >> product >> quantity;
        addProduct(warehouse, product, quantity);
        std::printf("OK\n");
    } else if (command == commands::INCREASE_QUANTITY) {
        std::string product;
        uint quantity;
        ss >> product >> quantity;
        increaseQuantity(warehouse, product, quantity);
        std::printf("OK\n");
    } else if (command == commands::ADD_ORDER) {
        std::string customer, product;
        uint quantity;
        ss >> customer >> product >> quantity;
        addOrder(warehouse, customer, product, quantity);
        std::printf("OK\n");
    } else if (command == commands::PROCESS_ORDERS) {
        processOrders(warehouse);
    } else if (command == commands::REPORT_PENDING) {
        reportPending(warehouse);
    } else if (command == commands::SOLD_QUANTITY) {
        std::string product;
        ss >> product;
        reportSoldQuantity(warehouse, product);
    } else if (command == commands::SHOW_STOCK) {
        showStock(warehouse);
    }
}

int main() {
    Warehouse warehouse;
    std::string line;
    while (std::getline(std::cin, line)) {
        processCommand(warehouse, line);
    }

    return 0;
}
